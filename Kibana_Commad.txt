2:- to update the tl pwhere exist more than one tlp 
@Mehlab Faryad  We can use this scripit and also scripit is updated when the tlp is adding.
POST /{index_name}/_update_by_query
{
  "script": {
    "source": """
      Set tlp = new HashSet();
            if (ctx._source.tlp != null) {
                for (t in ctx._source.tlp) {
                    tlp.add(t.toLowerCase());
                }
            }
            Map sorted_dict = ["white": 1, "green": 2, "amber": 3, "red": 4];
            Set valid_tlp = new HashSet();
            for (t in tlp) {
                if (sorted_dict.containsKey(t)) {
                    valid_tlp.add(t);
                }
            }
            if (!valid_tlp.isEmpty()) {
                List tlpList = new ArrayList(valid_tlp);
                tlpList.sort((a, b) -> sorted_dict.get(a) - sorted_dict.get(b));
                ctx._source.tlp = tlpList.subList(0, 1);
            } 
     """
  },
   "query": {
    "bool": {
      "must": [
        {
          "exists": { "field": "tlp" }
        }
      ],
      "filter": [
        {
          "range": {
            "tlp": {
              "gt": 1
            }
          }
        }
      ]
    }
  }
}
